# Kurashi Homes Demo – Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a 3-language (en/my/ja) Next.js demo of a Tokyo rental-apartment site for foreigners — home, search, and listing-detail pages over mock data — that the client can review on a preview link and that becomes the base of the production site.

**Architecture:** Next.js 16 App Router with a `[locale]` segment (next-intl). Pages call only an async data-access layer (`src/lib/data/index.ts`) that reads typed mock data today and will read Payload CMS in week 2. Domain logic (initial-cost calculation, filtering, URL⇄filter conversion, form validation) lives in pure, unit-tested modules under `src/lib/`.

**Tech Stack:** Next.js 16.4.6, React 19.3.1, TypeScript 5 (strict), Tailwind CSS 4.3.5, next-intl 4.14.10, zod 4.6.7, Vitest 5.1.0.

**Spec:** `docs/superpowers/specs/2026-11-11-demo-design.md`

## Global Constraints

- Next.js **16.4.6** exactly (Payload 3.93 supports `next >=16.3.3 <17`). Next 16 uses `src/proxy.ts`, not `middleware.ts`. `params` and `searchParams` are Promises.
- Locales: `en`, `my`, `ja`; prefix always (`/en`, `/my`, `/ja`); default `en`.
- Pages and components never import from `src/lib/data/mock/*` — only from `@/lib/data`.
- Rented listings (`status: 'rented'`) never appear in search results, featured, similar, or area counts — but their detail page still works.
- Brand name, colors, contact links live only in `src/config/site.ts`. Temporary brand: **Kurashi Homes**; colors ink `#1f2a44`, accent `#e4572e`, paper `#faf8f4`.
- Do not use any images or copy from arealty.jp. Photos come only from the verified Unsplash list in Task 5.
- Preview/staging must be noindex: `site.env !== 'production'` ⇒ `robots.txt` disallows all and pages emit `robots: noindex, nofollow`. Default env is `staging`.
- Burmese and Japanese copy written by Claude is a draft for the client's native speakers to review. Station/line/area names in `my` use the Latin (English) names.
- Shell: commands below are POSIX (Git Bash). Run from the project root `D:\Freelancer\real estate website`.
- Commit after every task with a `feat:`/`chore:`/`test:` message ending with the line `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`. Do **not** add `bao-gia/` or `chao-gia-vlance-94803.md`.

```ts
export interface Station { slug: string; name: LocalizedText; lineSlugs: string[]; areaSlug: string }

export const LAYOUTS = ['1R', '1K', '1DK', '1LDK', '2K', '2DK', '2LDK', '3LDK'] as const;
export type Layout = (typeof LAYOUTS)[number];

export const LISTING_STATUSES = ['available', 'applied', 'rented'] as const;
export type ListingStatus = (typeof LISTING_STATUSES)[number];

export const STORED_CONDITION_KEYS = [
  'foreignerOk', 'noGuarantor', 'studentOk', 'petOk', 'furnished', 'overseasApplication', 'autoLock',
] as const;
export type StoredConditionKey = (typeof STORED_CONDITION_KEYS)[number];

/** Stored flags plus values derived from other fields. */
export const CONDITION_KEYS = [...STORED_CONDITION_KEYS, 'noKeyMoney', 'newlyBuilt'] as const;
export type ConditionKey = (typeof CONDITION_KEYS)[number];

export const SORTS = ['newest', 'price-asc', 'price-desc'] as const;
export type SortKey = (typeof SORTS)[number];

export interface InitialCosts {
```

```ts
import type { Listing } from '@/lib/data/types';
import { monthlyTotal } from './derived';

export type InitialCostKey =
  | 'advanceRent' | 'deposit' | 'keyMoney' | 'brokerage'
  | 'guaranteeFee' | 'insurance' | 'keyExchange' | 'cleaning';

export interface InitialCostItem { key: InitialCostKey; amount: number; months?: number }

export function calcInitialCost(
  listing: Pick<Listing, 'rent' | 'managementFee' | 'initialCosts'>,
): { items: InitialCostItem[]; total: number } {
  const { rent, initialCosts: c } = listing;
  const byMonths = (key: InitialCostKey, months: number): InitialCostItem => ({
    key, amount: Math.round(rent * months), months,
  });
  const items: InitialCostItem[] = [
    { key: 'advanceRent', amount: monthlyTotal(listing) },
    byMonths('deposit', c.depositMonths),
    byMonths('keyMoney', c.keyMoneyMonths),
    byMonths('brokerage', c.brokerageMonths),
    { key: 'guaranteeFee', amount: c.guaranteeFee },
    { key: 'insurance', amount: c.insurance },
    { key: 'keyExchange', amount: c.keyExchange },
    { key: 'cleaning', amount: c.cleaning },
  ];
  return { items, total: items.reduce((sum, i) => sum + i.amount, 0) };
}
```

```ts
import { z } from 'zod';
import {
  CONDITION_KEYS, LAYOUTS, SORTS,
  type ConditionKey, type Layout, type Listing, type SortKey, type Station,
} from '@/lib/data/types';
import { hasCondition } from './derived';

export interface Filters {
  area?: string;
  line?: string;
  station?: string;
  maxWalk?: number;
  minRent?: number;
  maxRent?: number;
  layouts?: Layout[];
  conditions?: ConditionKey[];
  sort?: SortKey;
  page?: number;
}

export type SearchParamsInput = URLSearchParams | Record<string, string | string[] | undefined>;

const KEY_ORDER = ['area', 'line', 'station', 'maxWalk', 'minRent', 'maxRent', 'layouts', 'conditions', 'sort', 'page'] as const;

const slug = z.string().regex(/^[a-z0-9-]{1,40}$/).optional().catch(undefined);
const int = (min: number, max: number) => z.coerce.number().int().min(min).max(max).optional().catch(undefined);

const scalarSchema = z.object({
  area: slug,
  line: slug,
  station: slug,
  maxWalk: int(1, 60),
  minRent: int(0, 10_000_000),
  maxRent: int(0, 10_000_000),
  sort: z.enum(SORTS).optional().catch(undefined),
  page: int(1, 1000),
});

function readParam(input: SearchParamsInput, key: string): string | undefined {
  const raw = input instanceof URLSearchParams ? input.get(key) ?? undefined : input[key];
  const value = Array.isArray(raw) ? raw[0] : raw;
  return value === undefined || value === '' ? undefined : value;
}

function readList<T extends string>(value: string | undefined, allowed: readonly T[]): T[] | undefined {
  if (!value) return undefined;
  const picked = [...new Set(value.split(','))].filter((v): v is T => (allowed as readonly string[]).includes(v));
  return picked.length ? picked : undefined;
}

export function parseFilters(input: SearchParamsInput): Filters {
  const scalars = scalarSchema.parse({
    area: readParam(input, 'area'),
    line: readParam(input, 'line'),
    station: readParam(input, 'station'),
    maxWalk: readParam(input, 'maxWalk'),
    minRent: readParam(input, 'minRent'),
    maxRent: readParam(input, 'maxRent'),
    sort: readParam(input, 'sort'),
    page: readParam(input, 'page'),
  });
  const filters: Filters = {
    ...scalars,
    layouts: readList(readParam(input, 'layouts'), LAYOUTS),
    conditions: readList(readParam(input, 'conditions'), CONDITION_KEYS),
  };
  if (filters.minRent !== undefined && filters.maxRent !== undefined && filters.minRent > filters.maxRent) {
    [filters.minRent, filters.maxRent] = [filters.maxRent, filters.minRent];
  }
  return Object.fromEntries(Object.entries(filters).filter(([, v]) => v !== undefined)) as Filters;
}

export function serializeFilters(filters: Filters): string {
  const params = new URLSearchParams();
  for (const key of KEY_ORDER) {
    const value = filters[key];
    if (value === undefined) continue;
    if (Array.isArray(value)) {
      if (value.length) params.set(key, value.join(','));
    } else if (!(key === 'page' && value === 1)) {
      params.set(key, String(value));
    }
  }
  return params.toString();
}

const SORTERS: Record<SortKey, (a: Listing, b: Listing) => number> = {
  newest: (a, b) => b.publishedAt.localeCompare(a.publishedAt) || a.code.localeCompare(b.code),
  'price-asc': (a, b) => a.rent - b.rent || a.code.localeCompare(b.code),
  'price-desc': (a, b) => b.rent - a.rent || a.code.localeCompare(b.code),
};

export function filterListings(
  listings: Listing[],
  filters: Filters,
  ctx: { stations: Station[]; now?: Date },
): Listing[] {
  const now = ctx.now ?? new Date();
  const stationLines = new Map(ctx.stations.map((s) => [s.slug, s.lineSlugs]));
  const needsStation = filters.station !== undefined || filters.line !== undefined || filters.maxWalk !== undefined;

  const matches = (l: Listing): boolean => {
    if (filters.area && l.areaSlug !== filters.area) return false;
    if (needsStation) {
      const ok = l.stations.some((s) =>
        (!filters.station || s.stationSlug === filters.station) &&
        (!filters.line || (stationLines.get(s.stationSlug) ?? []).includes(filters.line)) &&
        (filters.maxWalk === undefined || s.walkMinutes <= filters.maxWalk));
      if (!ok) return false;
    }
    if (filters.minRent !== undefined && l.rent < filters.minRent) return false;
    if (filters.maxRent !== undefined && l.rent > filters.maxRent) return false;
    if (filters.layouts?.length && !filters.layouts.includes(l.layout)) return false;
    if (filters.conditions?.some((c) => !hasCondition(l, c, now))) return false;
    return true;
  };

  return listings.filter(matches).sort(SORTERS[filters.sort ?? 'newest']);
}
```

```ts
import { filterListings, type Filters } from '@/lib/listing/filters';
import { AREAS } from './mock/areas';
import { LINES } from './mock/lines';
import { LISTINGS } from './mock/listings';
import { STATIONS } from './mock/stations';
import type { Area, Line, Listing, ListingPage, Station } from './types';

export const PAGE_SIZE = 12;

const isOpen = (l: Listing) => l.status !== 'rented';

export async function getListings(filters: Filters, now: Date = new Date()): Promise<ListingPage> {
  const matched = filterListings(LISTINGS.filter(isOpen), filters, { stations: STATIONS, now });
  const total = matched.length;
  const pageCount = Math.max(1, Math.ceil(total / PAGE_SIZE));
  const page = Math.min(Math.max(filters.page ?? 1, 1), pageCount);
  return { items: matched.slice((page - 1) * PAGE_SIZE, page * PAGE_SIZE), total, page, pageCount };
}

export async function getListing(code: string): Promise<Listing | null> {
  return LISTINGS.find((l) => l.code === code.toUpperCase()) ?? null;
}

export async function getSimilarListings(listing: Listing, limit = 4): Promise<Listing[]> {
  const stationSet = new Set(listing.stations.map((s) => s.stationSlug));
  const score = (l: Listing) =>
    (l.areaSlug === listing.areaSlug ? 2 : 0) +
    (l.stations.some((s) => stationSet.has(s.stationSlug)) ? 3 : 0) +
    (Math.abs(l.rent - listing.rent) <= listing.rent * 0.2 ? 1 : 0);
  return LISTINGS
    .filter((l) => isOpen(l) && l.code !== listing.code)
    .map((l) => ({ l, s: score(l) }))
    .filter(({ s }) => s > 0)
    .sort((a, b) => b.s - a.s || Math.abs(a.l.rent - listing.rent) - Math.abs(b.l.rent - listing.rent))
    .slice(0, limit)
    .map(({ l }) => l);
}

export async function getFeaturedListings(limit = 6): Promise<Listing[]> {
  return LISTINGS.filter((l) => l.featured && isOpen(l)).slice(0, limit);
}

export async function getAreas(): Promise<Area[]> {
  return AREAS;
}

export async function getLines(): Promise<Line[]> {
  return LINES;
}

export async function getStations(lineSlug?: string): Promise<Station[]> {
  return lineSlug ? STATIONS.filter((s) => s.lineSlugs.includes(lineSlug)) : STATIONS;
}

export async function countListingsByArea(): Promise<Record<string, number>> {
  const counts: Record<string, number> = Object.fromEntries(AREAS.map((a) => [a.slug, 0]));
  for (const l of LISTINGS) if (isOpen(l)) counts[l.areaSlug] = (counts[l.areaSlug] ?? 0) + 1;
  return counts;
}

export async function getListingRefs(): Promise<{ code: string; areaSlug: string; slug: string }[]> {
  return LISTINGS.map(({ code, areaSlug, slug }) => ({ code, areaSlug, slug }));
}
```

`src/i18n/merge.ts`:
```ts
export type Messages = { [key: string]: string | Messages };

export function mergeMessages(base: Messages, override: Messages): Messages {
  const out: Messages = { ...base };
  for (const [key, value] of Object.entries(override)) {
    const baseValue = base[key];
    if (value === '') continue;
    out[key] = typeof value === 'object' && typeof baseValue === 'object'
      ? mergeMessages(baseValue, value)
      : value;
  }
  return out;
}
```

```ts
export function buildAlternates(locale: Locale, path = '') {

  for (const l of LOCALES) languages[l] = localeUrl(l, path);
```

```ts
import { z } from 'zod';

type Field = 'name' | 'email' | 'message';
type ErrorCode = 'required' | 'invalidEmail';

export function validateInquiry(input: { name: string; email: string; phone: string; message: string }):
  { ok: true } | { ok: false; errors: Partial<Record<Field, ErrorCode>> } {
  const errors: Partial<Record<Field, ErrorCode>> = {};
  if (!input.name.trim()) errors.name = 'required';
  if (!input.email.trim()) errors.email = 'required';
  else if (!z.email().safeParse(input.email.trim()).success) errors.email = 'invalidEmail';
  if (!input.message.trim()) errors.message = 'required';
  return Object.keys(errors).length ? { ok: false, errors } : { ok: true };
}
```

```ts
export type SearchOption = { value: string; label: string };
export type StationOption = SearchOption & { lines: string[] };
export interface SearchOptions { areas: SearchOption[]; lines: SearchOption[]; stations: StationOption[] }

export async function buildSearchOptions(locale: Locale): Promise<SearchOptions> {
  const [areas, lines, stations] = await Promise.all([getAreas(), getLines(), getStations()]);

    stations: stations
      .map((s) => ({ value: s.slug, label: pickText(s.name, locale), lines: s.lineSlugs }))
      .sort((a, b) => a.label.localeCompare(b.label, locale)),
  };
}
```

```tsx
export function Pagination({ filters, page, pageCount }: { filters: Filters; page: number; pageCount: number }) {

  );
```

```tsx
    description: pickText(listing.description, l).slice(0, 155),
```
